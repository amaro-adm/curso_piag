Scikit-learn — Modelo de Classificação

Treinamento do modelo supervisionado com dados históricos de 15.000 inspeções para prever peças defeituosas.

# ============================================================

# ETAPA 1: MODELO DE CLASSIFICAÇÃO COM SCIKIT-LEARN

# Objetivo: Treinar um classificador para prever defeitos

# em rolamentos a partir de medições dos sensores.

# ============================================================



import pandas as pd

from sklearn.model_selection import train_test_split

from sklearn.ensemble import RandomForestClassifier

from sklearn.metrics import classification_report, confusion_matrix

from sklearn.preprocessing import StandardScaler



# 1. Carregar os dados históricos de inspeção

df = pd.read_csv("inspecoes_rolamentos.csv")



# 2. Separar features (medições) do rótulo (resultado)

# Features: diametro_mm, espessura_mm, temp_tempera_C, vibracao_hz, rugosidade_Ra

# Target: resultado (0=aprovado, 1=defeituoso)

X = df[["diametro_mm", "espessura_mm", "temp_tempera_C",

         "vibracao_hz", "rugosidade_Ra"]]

y = df["resultado"]



# 3. Normalizar os dados — importante pois as escalas variam muito

# (diâmetro ~47mm vs temperatura ~850°C)

scaler = StandardScaler()

X_scaled = scaler.fit_transform(X)



# 4. Dividir em treino (80%) e teste (20%)

X_train, X_test, y_train, y_test = train_test_split(

    X_scaled, y, test_size=0.2, random_state=42, stratify=y

)



# 5. Treinar Random Forest — ideal para dados tabulares industriais

# n_estimators=200: usa 200 árvores de decisão em paralelo

# class_weight='balanced': compensa o desbalanceamento (96.8% aprovadas vs 3.2% defeituosas)

modelo = RandomForestClassifier(

    n_estimators=200,

    class_weight="balanced",

    random_state=42

)

modelo.fit(X_train, y_train)



# 6. Avaliar o modelo no conjunto de teste

y_pred = modelo.predict(X_test)

print(classification_report(y_test, y_pred))

print(confusion_matrix(y_test, y_pred))



# 7. Predição em tempo real — nova peça do sensor

nova_peca = [[47.02, 14.01, 855, 11.3, 0.42]]

nova_peca_scaled = scaler.transform(nova_peca)

resultado = modelo.predict(nova_peca_scaled)

# resultado[0] == 0 → APROVADA | resultado[0] == 1 → DEFEITUOSA

ETAPA 2

TensorFlow — Rede Neural Profunda

Construção de uma rede neural densa de baixo nível com TensorFlow puro para capturar padrões não-lineares complexos nos dados de inspeção.

# ============================================================

# ETAPA 2: REDE NEURAL COM TENSORFLOW (BAIXO NÍVEL)

# Objetivo: Usar Deep Learning para capturar relações

# não-lineares entre as medições dos sensores.

# ============================================================



import tensorflow as tf

import numpy as np



# 1. Converter dados para tensores TensorFlow

# TensorFlow opera com tensores (arrays multidimensionais otimizados para GPU)

X_train_tf = tf.constant(X_train, dtype=tf.float32)

y_train_tf = tf.constant(y_train.values, dtype=tf.float32)



# 2. Definir variáveis treináveis (pesos e bias) manualmente

# Camada 1: 5 inputs → 64 neurônios

W1 = tf.Variable(tf.random.normal([5, 64], stddev=0.1))

b1 = tf.Variable(tf.zeros([64]))



# Camada 2: 64 → 32 neurônios

W2 = tf.Variable(tf.random.normal([64, 32], stddev=0.1))

b2 = tf.Variable(tf.zeros([32]))



# Camada de saída: 32 → 1 (probabilidade de defeito)

W_out = tf.Variable(tf.random.normal([32, 1], stddev=0.1))

b_out = tf.Variable(tf.zeros([1]))



# 3. Função de forward pass (propagação direta)

def forward(X):

    z1 = tf.nn.relu(tf.matmul(X, W1) + b1)  # Ativação ReLU

    z2 = tf.nn.relu(tf.matmul(z1, W2) + b2)

    output = tf.nn.sigmoid(tf.matmul(z2, W_out) + b_out)  # Sigmoid → [0,1]

    return output



# 4. Função de perda: Binary Cross-Entropy

# Penaliza previsões erradas proporcionalmente à confiança do modelo

def compute_loss(y_true, y_pred):

    return tf.reduce_mean(

        tf.nn.sigmoid_cross_entropy_with_logits(

            labels=y_true, logits=y_pred

        )

    )



# 5. Otimizador Adam — ajusta pesos via gradiente descendente adaptativo

optimizer = tf.optimizers.Adam(learning_rate=0.001)



# 6. Loop de treinamento manual com GradientTape

for epoch in range(100):

    with tf.GradientTape() as tape:

        predictions = forward(X_train_tf)

        loss = compute_loss(y_train_tf, predictions)

    # Calcular gradientes e atualizar pesos

    grads = tape.gradient(loss, [W1,b1,W2,b2,W_out,b_out])

    optimizer.apply_gradients(zip(grads, [W1,b1,W2,b2,W_out,b_out]))

ETAPA 3

Keras — API de Alto Nível

Mesmo modelo da Etapa 2, mas com código 5x mais simples usando Keras sobre TensorFlow. Ideal para prototipagem rápida.

# ============================================================

# ETAPA 3: MESMA REDE NEURAL COM KERAS (ALTO NÍVEL)

# Objetivo: Simplificar a construção do modelo Deep Learning.

# Keras roda "por cima" do TensorFlow, abstraindo a complexidade.

# ============================================================



from tensorflow import keras

from tensorflow.keras import layers



# 1. Construir modelo sequencial (camadas empilhadas)

modelo_keras = keras.Sequential([

    # Camada de entrada: recebe 5 features dos sensores

    layers.Input(shape=(5,)),



    # Camada densa 1: 64 neurônios com ReLU

    # ReLU: se valor > 0, mantém; se < 0, zera (introduz não-linearidade)

    layers.Dense(64, activation="relu"),

    layers.Dropout(0.3),  # Desliga 30% dos neurônios aleatoriamente (evita overfitting)



    # Camada densa 2: 32 neurônios

    layers.Dense(32, activation="relu"),

    layers.Dropout(0.2),



    # Camada de saída: 1 neurônio com Sigmoid → probabilidade [0, 1]

    layers.Dense(1, activation="sigmoid")

])



# 2. Compilar: definir perda, otimizador e métricas

modelo_keras.compile(

    optimizer="adam",

    loss="binary_crossentropy",     # Perda para classificação binária

    metrics=["accuracy", "recall"]  # Recall é crucial: não podemos perder defeitos!

)



# 3. Treinar com early stopping (para quando não melhora mais)

early_stop = keras.callbacks.EarlyStopping(

    patience=10, restore_best_weights=True

)



historico = modelo_keras.fit(

    X_train, y_train,

    epochs=100,

    batch_size=32,

    validation_split=0.15,     # 15% para validação interna

    callbacks=[early_stop],

    class_weight={0: 1.0, 1: 30.0}  # Peso 30x para defeituosas (desbalanceamento)

)



# 4. Salvar modelo treinado para deploy na linha de produção

modelo_keras.save("modelo_rolamentos_v1.h5")

OpenCV — Inspeção Visual

Complemento visual: câmera na linha captura imagens da superfície do rolamento. OpenCV detecta trincas, arranhões e imperfeições que sensores numéricos não captam.

# ============================================================

# ETAPA 4: INSPEÇÃO VISUAL COM OPENCV

# Objetivo: Detectar defeitos visuais (trincas, arranhões)

# na superfície dos rolamentos usando processamento de imagem.

# ============================================================



import cv2

import numpy as np



# 1. Capturar frame da câmera industrial na linha de produção

camera = cv2.VideoCapture(0)  # Câmera ID 0 na estação de inspeção

ret, frame = camera.read()



# 2. Pré-processamento: converter para escala de cinza

# Imagens em cinza são mais rápidas de processar e eliminam ruído de cor

cinza = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)



# 3. Aplicar filtro Gaussiano para suavizar ruído do sensor

suavizada = cv2.GaussianBlur(cinza, (5, 5), 0)



# 4. Detecção de bordas com Canny — encontra contornos de defeitos

# Limiar baixo (50) e alto (150): sensibilidade calibrada para trincas

bordas = cv2.Canny(suavizada, 50, 150)



# 5. Encontrar contornos (possíveis defeitos)

contornos, _ = cv2.findContours(

    bordas, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE

)



# 6. Filtrar contornos por área — eliminar ruído (contornos minúsculos)

defeitos_detectados = []

for contorno in contornos:

    area = cv2.contourArea(contorno)

    if area > 100:  # Área mínima: 100 pixels² (calibrado empiricamente)

        defeitos_detectados.append(contorno)

        # Desenhar retângulo vermelho ao redor do defeito

        x, y, w, h = cv2.boundingRect(contorno)

        cv2.rectangle(frame, (x,y), (x+w,y+h), (0,0,255), 2)



# 7. Decisão: se detectou defeitos visuais → rejeitar peça

tem_defeito_visual = len(defeitos_detectados) > 0

print(f"Defeitos visuais encontrados: {len(defeitos_detectados)}")



camera.release()

    nome_classe = ["ok","trinca","corrosao","deformacao","arranhao"][int(classe)]

    print(f"Defeito: {nome_classe} | Confiança: {confianca:.1%}")



# 5. Decisão automática na linha de produção

tem_defeito = any(r.cls.item() != 0 for r in resultados[0].boxes)

# ETAPA 6: ANÁLISE DE RELATÓRIOS COM NLTK

# Objetivo: Processar textos escritos por inspetores para

# ============================================================



import nltk

from nltk.tokenize import word_tokenize

from nltk.corpus import stopwords

from nltk.sentiment import SentimentIntensityAnalyzer

from collections import Counter



# 1. Relatório de exemplo escrito pelo inspetor

relatorio = """Peça lote 4523 apresentou trinca severa na pista interna.

Vibração muito acima do normal durante teste rotativo.

Temperatura de têmpera possivelmente fora do padrão.

Recomendo descarte imediato e revisão do forno 3."""



# 2. Tokenização: quebrar texto em palavras individuais

tokens = word_tokenize(relatorio, language="portuguese")

# Resultado: ['Peça', 'lote', '4523', 'apresentou', 'trinca', ...]



# 3. Remover stopwords (palavras sem significado: "de", "na", "do")

stop_pt = set(stopwords.words("portuguese"))

palavras_relevantes = [t.lower() for t in tokens

                       if t.isalpha() and t.lower() not in stop_pt]



# 4. Frequência de termos — quais palavras aparecem mais?

frequencia = Counter(palavras_relevantes)

print("Termos mais frequentes:", frequencia.most_common(5))



# 5. Detecção de palavras-chave de gravidade

palavras_criticas = {"severa", "descarte", "imediato", "grave", "urgente"}

gravidade = len(palavras_criticas.intersection(set(palavras_relevantes)))

print(f"Nível de gravidade: {gravidade}/5 palavras críticas detectadas")



# 6. POS Tagging: identificar classes gramaticais

# Útil para extrair entidades como nomes de equipamentos

tagged = nltk.pos_tag(tokens)

# Resultado: [('trinca', 'NN'), ('severa', 'JJ'), ...]

spaCy — Extração de Entidades em Produção

Versão industrial do processamento de texto: spaCy extrai automaticamente lotes, equipamentos e tipos de defeito de milhares de relatórios por dia.

# ============================================================

# ETAPA 7: EXTRAÇÃO DE ENTIDADES COM spaCy (PRODUÇÃO)

# Objetivo: Processar milhares de relatórios em larga escala,

# extraindo entidades nomeadas (lotes, equipamentos, defeitos).

# ============================================================



import spacy

from spacy.matcher import Matcher



# 1. Carregar modelo pré-treinado em português (inclui NER + parsing)

nlp = spacy.load("pt_core_news_lg")  # Modelo grande: melhor precisão



# 2. Criar matcher customizado para termos da fábrica

matcher = Matcher(nlp.vocab)



# Padrão: detectar referências a equipamentos (ex: "forno 3", "prensa 7")

padrao_equipamento = [{"LOWER": {"IN": ["forno", "prensa", "torno"]}},

                      {"IS_DIGIT": True}]

matcher.add("EQUIPAMENTO", [padrao_equipamento])



# Padrão: detectar códigos de lote (ex: "lote 4523")

padrao_lote = [{"LOWER": "lote"}, {"IS_DIGIT": True}]

matcher.add("LOTE", [padrao_lote])



# 3. Processar relatório (pipeline completo: tokenize → parse → NER)

relatorio = """Peça lote 4523 apresentou trinca severa na pista interna.

Temperatura do forno 3 estava 40 graus acima do limite."""



doc = nlp(relatorio)



# 4. Extrair entidades nomeadas automaticamente

print("=== Entidades Detectadas ===")

for ent in doc.ents:

    print(f"  {ent.text} → {ent.label_}")



# 5. Aplicar matcher customizado

matches = matcher(doc)

for match_id, start, end in matches:

    nome_regra = nlp.vocab.strings[match_id]

    span = doc[start:end]

    print(f"  [{nome_regra}] → {span.text}")

    # Saída: [LOTE] → lote 4523  |  [EQUIPAMENTO] → forno 3



# 6. Análise de dependência sintática (quem fez o quê?)

for token in doc:

    if token.dep_ == "nsubj":  # Sujeito da frase

        print(f"  Sujeito: {token.text} → Verbo: {token.head.text}")
